Flutter Example (Dart):

                
import 'dart:convert';
import 'package:http/http.dart' as http;

Future<Map<String, dynamic>> extractSheinCart(String sheinLink) async {
  final url = Uri.parse('https://veloxshoppingiq.com/api/shein_extract.php');
  
  final response = await http.post(
    url,
    headers: {'Content-Type': 'application/json'},
    body: jsonEncode({'shein_link': sheinLink}),
  );
  
  if (response.statusCode == 200) {
    return jsonDecode(response.body);
  } else {
    throw Exception('Failed to extract cart');
  }
}

// Usage:
try {
  var result = await extractSheinCart('YOUR_SHEIN_LINK');
  if (result['success']) {
    print('Items: ${result['items']}');
    print('Total: \$${result['total_price']}');
  } else {
    print('Error: ${result['message']}');
  }
} catch (e) {
  print('Exception: $e');
}
