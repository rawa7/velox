API Documentation

Response:

  "data": [

      "product_image": "https://veloxshoppingiq.com/uploads/6396.PNG"
    }
  ]
}

➕ Create Banner

POST /api/shop_banners.php
Content-Type: multipart/form-data

Parameters:
- title (required): Banner title
- title_kurdish: Title in Kurdish
- title_arabic: Title in Arabic
- description: Banner description
- description_kurdish: Description in Kurdish
- description_arabic: Description in Arabic
- product_id (required): Product ID to link
- banner_image: Banner image file
- position: Display order (default: 0)
- is_active: 1 or 0 (default: 1)

🔄 Update Banner

PUT /api/shop_banners.php
Content-Type: application/x-www-form-urlencoded

Parameters: (same as POST + banner_id)

🗑️ Delete Banner

DELETE /api/shop_banners.php
Content-Type: application/json

{
  "banner_id": 1
}

🎯 Flutter Example

// Fetch banners
final response = await http.get(
  Uri.parse('https://veloxshoppingiq.com/api/shop_banners.php')
);

final data = json.decode(response.body);
List banners = data['data'];

// Display in carousel
CarouselSlider(
  items: banners.map((banner) {
    return GestureDetector(
      onTap: () {
        // Navigate to product page
        Navigator.push(
          context,
          MaterialPageRoute(
            builder: (context) => ProductPage(
              productId: banner['product_id']
            )
          )
        );
      },
      child: Image.network(
        banner['banner_image'] ?? banner['product_image'],
        fit: BoxFit.cover,
      ),
    );
  }).toList(),
)
